name: Java 24 Maven CI

on:
  push:
    branches:
      - '**'  # All branches
  pull_request:
    branches:
      - '**'

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout source code
        uses: actions/checkout@v4

      - name: Set up JDK 24
        uses: actions/setup-java@v4
        with:
          distribution: 'temurin'  # Eclipse Temurin is widely used and reliable
          java-version: '24'       # ✅ Java 24

      - name: Build and run tests with Maven
        run: mvn clean install -B
